public class Util {

	public SSLSocketFactory validateCertificate(Logger logger) {

		SSLSocketFactory sslSocketFactory =null;
		try {
			SSLContext context = SSLContext.getInstance("TLSv1.2");
			logger.debug( "SSLContext created");
//			ClassPathResource resource = new ClassPathResource("ADCBBANK.jks");
//			File file = resource.getFile();
//			KeyStore ks = KeyStore.getInstance("jks");
//			FileInputStream fis = new FileInputStream(file);
//			ks.load(fis, "password".toCharArray());
			KeyStore ks = CertLoader.getKeyStore();
			char[] password = CertLoader.getPassword();
			KeyManagerFactory kmf = KeyManagerFactory.getInstance("SunX509");
			kmf.init(ks, password); // new code
//			kmf.init(ks, "password".toCharArray());
			context.init(kmf.getKeyManagers(), null, null);
			sslSocketFactory = context.getSocketFactory();
		}catch(Exception e) {
			StringWriter str = new StringWriter();
			e.printStackTrace(new PrintWriter(str));
			logger.error(str.toString());
		}

		return sslSocketFactory;

	}
